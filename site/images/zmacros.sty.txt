\DeclareRobustCommand{\subsetsim}{\mathrel{\mathchoice
  {\subsetsim@{\displaystyle}{\scriptstyle}{.47\fontdimen5\textfont2}{1.12}{.72}{.050em}{.23\fontdimen5\textfont2}}%
  {\subsetsim@{\textstyle}{\scriptstyle}{.47\fontdimen5\textfont2}{1.12}{.72}{.050em}{.23\fontdimen5\textfont2}}%
  {\subsetsim@{\scriptstyle}{\scriptscriptstyle}{.41\fontdimen5\scriptfont2}{1.12}{.72}{.035em}{.20\fontdimen5\scriptfont2}}%
  {\subsetsim@{\scriptscriptstyle}{\scriptscriptstyle}{.53\fontdimen5\scriptscriptfont2}{1.06}{.70}{.025em}{.22\fontdimen5\scriptscriptfont2}}%
}}
\newcommand{\subsetsim@}[7]{%
  \begingroup
  \setbox0=\hbox{$\m@th#1\subseteq$}% reference box: same width/height/depth as \subseteq
  \setbox2=\hbox{$\m@th#1\subset*$}% visible subset glyph
  \setbox4=\hbox{\scalebox{#4}[#5]{%
    \ooalign{%
      \hfil$\m@th#2\sim$\hfil\cr
      \hfil\kern.025em$\m@th#2\sim$\hfil\cr
    }%
  }}% visible squiggle
  \setbox6=\hbox{%
    \ooalign{%
      \hbox to \wd0{\hss\raise#7\copy2\hss}\cr
      \hbox to \wd0{\hss\kern#6\lower#3\copy4\hss}\cr
    }%
  }%
  \ht6=\ht0 % prevent extra height
  \dp6=\dp0 % prevent extra depth/line padding
  \box6
  \endgroup
}

  % #1 = env name, e.g. lemma
  % #2 = printed heading / uppercase cref singular, e.g. Lemma

}
